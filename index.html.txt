function normalizeSliderData(raw){
  if(!raw) return [];
  if(Array.isArray(raw)) return raw.map((v,i)=>({id:String(i),...(typeof v==='object'&&v?v:{image:v})}));
  if(typeof raw==='object') {
    if(sliderImageValue(raw)) return [{id:'single',...raw}];

}
